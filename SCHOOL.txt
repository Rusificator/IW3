7. Описать класс "Школа, ученики". Предусмотреть возможность работы с произвольным числом учеников, поиска ученика по заданному классу и фамилии, добавления учеников в школу, удаления из неё, сортировки учеников по разным полям. Написать программу, демонстрирующую работу с этим классом.



Код класса Student:

import java.util.*;

class St {
    private String fio;
    private String grade; // Класс
    private String addr;
    private int[] marks;

     St(String fio, String grade, String addr, int[] marks)
     {
        this.fio = fio;
        this.grade = grade;
        this.addr = addr;
        this.marks = marks;
    }

     double getAverage()
    {
        if (marks.length == 0) return 0;
        double sum = 0;
        for (int mark : marks)
        {
            sum += mark;
        }
        return sum / marks.length;
    }


     String getFio()
     {
         return fio;
     }
     String getGrade()
     {
         return grade;
     }
     String getAddress()
     {
         return addr;
     }
     int[] getMarks()
     {
         return marks;
     }


    public String toString()
    {
        return "\nУченик: "+fio +"\nКласс:"+grade+"\nАдрес:"+ addr +"\nОценки:"+Arrays.toString(marks) +"\nСредний балл:"+getAverage();

    }


    public static  Comparator <St> FioComparator=new Comparator<St>()
    {

        public int compare(St p1, St p2)
        {
            return p1.fio.compareToIgnoreCase(p2.fio);
        }
    };


        public static  Comparator <St> GradeComparator=new Comparator<St>()
        {
        public int compare(St p1, St p2)
        {
            int gradeCompare = p1.grade.compareToIgnoreCase(p2.grade);
            if (gradeCompare != 0) return gradeCompare;
            return p1.fio.compareToIgnoreCase(p2.fio);
        }
    };
    public static  Comparator <St> AverageComparator=new Comparator<St>()
    {
        public int compare(St p1, St p2)
        {
            return Double.compare(p2.getAverage(), p1.getAverage()); // По убыванию
        }
    };

}


Код класса School:

import java.util.*;
class School {
    private List<St> st = new ArrayList<>();


    public void add(St s)
    {
        st.add(s);
        System.out.println("Добавлен: " + s.getFio());
    }


    public void del(String fio) {
        int p=-1;
        for(int i=0;i<st.size();i++)
            if(fio.equalsIgnoreCase(st.get(i).getFio())) p=i;
        if(p>-1)
        {
            st.remove(p);
            System.out.println("Удалили ученика: "+fio);
        }
        else System.out.println("Такого ученика нет");
    }








    // Поиск по классу и фамилии
    public void find(String grade, String fio)
    {
        boolean f = false;
        for (St p : st) {
            if (p.getGrade().equalsIgnoreCase(grade) &&
                    p.getFio().toLowerCase().contains(fio.toLowerCase()))
            {
                System.out.println("Найден: " + p);
                f = true;
            }
        }
        if (!f) System.out.println("Ученик не найден");
    }


    public void sortByFio() {
        st.sort(St.FioComparator);
    }

    public void sortByGrade() {
        st.sort(St.GradeComparator);
    }

    public void sortByAverage()
    {
        st.sort(St.AverageComparator);
    }


    public void printAll()
    {
        if (st.isEmpty())
        {
            System.out.println("В школе нет учеников");
            return;
        }
        System.out.println("\nСписок учеников:");
        for (St s : st) {
            System.out.println(s);
        }
    }
}
